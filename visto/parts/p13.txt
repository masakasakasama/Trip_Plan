
const FLAG_FALLBACK={Australia:'🇦🇺',Philippines:'🇵🇭',Japan:'🇯🇵',Germany:'🇩🇪',France:'🇫🇷',Netherlands:'🇳🇱',Belgium:'🇧🇪',Poland:'🇵🇱','Türkiye':'🇹🇷',Turkey:'🇹🇷',Thailand:'🇹🇭',Vietnam:'🇻🇳',China:'🇨🇳','South Korea':'🇰🇷',Korea:'🇰🇷','Hong Kong':'🇭🇰','United States':'🇺🇸',USA:'🇺🇸','United States of America':'🇺🇸',Italy:'🇮🇹'};
function countryFlag(c){const name=String(c||'').trim();const id=NAME_TO_ID[name.toLowerCase()];return COUNTRY_META[id]?.flag||FLAG_FALLBACK[name]||'🌐'}
function flagsForTrip(t){return (t.countries||inferCountries(t)).map(countryFlag).filter(Boolean)}
function tripYearLabel(t){const m=String(t.startDate||tripDateLabel(t)||'').match(/20\d{2}/);return m?m[0]:'TRIP'}
function tripPrimaryLabel(t){return t.destination||getCities(t)[0]||t.title||'Trip'}
function flagStackHtml(flags,klass=''){return `<div class="trip-flag-stack ${klass}">${flags.slice(0,4).map(f=>`<span class="trip-flag">${f}</span>`).join('')}</div>`}
function statusLabel(t){const phase=typeof tripPhase==='function'?tripPhase(t):'';if(phase==='current')return'旅行中';if(phase==='upcoming')return'予定';if(phase==='past')return'過去';return t.status||'Trip'}
