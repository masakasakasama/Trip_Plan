function S15_haversine(a,b){
  const R=6371,toRad=v=>v*Math.PI/180;
  const dLat=toRad(b.lat-a.lat),dLon=toRad(b.lng-a.lng);
  const x=Math.sin(dLat/2)**2+Math.cos(toRad(a.lat))*Math.cos(toRad(b.lat))*Math.sin(dLon/2)**2;
  return 2*R*Math.asin(Math.sqrt(x));
}
function S15_flightStats(){
  const seed=S15_FLIGHT(),airports=seed.airports||{},flights=(seed.flights||[]).filter(f=>f.status==='flown');
  const airline={},airport={},decorated=[];let total=0;
  flights.forEach(f=>{
    const a=airports[f.from],b=airports[f.to];
    const km=a&&b?S15_haversine(a,b):0; total+=km;
    const ar=airline[f.airline]||(airline[f.airline]={flights:0,km:0});ar.flights++;ar.km+=km;
    [f.from,f.to].forEach(code=>{airport[code]=(airport[code]||0)+1});
    decorated.push({...f,km});
  });
  return {
    flights:decorated,totalKm:Math.round(total),
    earthLaps:total/Number(seed.earthCircumferenceKm||40075),
    airline,airport,
    airlineCount:Object.keys(airline).length,
    airportCount:Object.keys(airport).length
  };
}
function S15_tripMatches(t){
  const q=S15_tripExplorer.query.trim().toLowerCase();
  const countries=(t.countries||inferCountries(t)).filter(Boolean),cities=getCities(t);
  if(q){
    const hay=[t.title,t.destination,...countries,...cities].filter(Boolean).join(' ').toLowerCase();
    if(!hay.includes(q))return false;
  }
  if(S15_tripExplorer.country&&!countries.includes(S15_tripExplorer.country))return false;
  if(S15_tripExplorer.year&&String(t.startDate||'').slice(0,4)!==S15_tripExplorer.year)return false;
  if(S15_tripExplorer.phase!=='all'&&tripPhase(t)!==S15_tripExplorer.phase)return false;
  return true;
}

}

function S15_qualityIssues(){
  const issues=[],all=trips(),ids=new Set();
  all.forEach(t=>{
    if(!t.id)issues.push({level:'error',text:`IDなし: ${t.title||'Untitled'}`});
    else if(ids.has(t.id))issues.push({level:'error',text:`Trip ID重複: ${t.id}`}); else ids.add(t.id);
    if(t.startDate&&t.endDate&&t.startDate>t.endDate)issues.push({level:'error',text:`日付逆転: ${t.title}`});
    if(!(t.countries||inferCountries(t)).filter(Boolean).length)issues.push({level:'warn',text:`国未設定: ${t.title}`});
    if(t.startDate&&t.endDate&&S15_exactDateKnown(t)){
      (t.days||[]).forEach(d=>{if(d.date&&(d.date<t.startDate||d.date>t.endDate))issues.push({level:'warn',text:`Trip期間外の予定: ${t.title} ${d.date}`})});
    }
    const dayIds=new Set();
    (t.days||[]).forEach(d=>{
      if(d.id&&dayIds.has(d.id))issues.push({level:'error',text:`Day ID重複: ${t.title} ${d.id}`}); if(d.id)dayIds.add(d.id);
      const times=(d.items||[]).map(i=>String(i.time||'')).filter(Boolean);
      if(times.some((v,i)=>i&&times[i-1]>v))issues.push({level:'warn',text:`時刻順不整合: ${t.title} ${d.date}`});
    });
  });
  const seed=S15_FLIGHT(),flightIds=new Set();
  (seed.flights||[]).forEach(f=>{
    if(flightIds.has(f.id))issues.push({level:'error',text:`Flight ID重複: ${f.id}`});flightIds.add(f.id);
    if(!seed.airports?.[f.from]||!seed.airports?.[f.to])issues.push({level:'error',text:`空港座標なし: ${f.flight||f.id} ${f.from}-${f.to}`});
  });
  const expected=new Set((typeof historicalTrips==='function'?historicalTrips():all.filter(t=>tripPhase(t)!=='upcoming')).flatMap(t=>(t.countries||inferCountries(t)).filter(Boolean)));
  const actual=new Set(getCountries());
  [...actual].filter(c=>!expected.has(c)).forEach(c=>issues.push({level:'error',text:`未来Tripが訪問済みに混入: ${c}`}));
  return issues;
}
function S15_qualityHtml(){
  const issues=S15_qualityIssues(),errors=issues.filter(x=>x.level==='error').length,warns=issues.filter(x=>x.level==='warn').length;
  return `<details class="quality-panel" ${errors?'open':''}><summary><span>データ品質</span><b class="${errors?'bad':'good'}">${errors?`${errors} errors`:warns?`${warns} warnings`:'OK'}</b></summary><div class="quality-body">${issues.map(x=>`<div class="quality-row ${x.level}"><span>${x.level==='error'?'×':'!'}</span><span>${esc(x.text)}</span></div>`).join('')||'<div class="quality-ok">日付、重複、集計、Flight参照の整合性チェック済み</div>'}</div></details>`;
}

function S15_monthIntersects(t,year,month){
  if(!t.startDate&&!t.endDate)return false;
  const start=t.startDate||t.endDate,end=t.endDate||t.startDate;
  const ms=`${year}-${String(month).padStart(2,'0')}-01`;
  const last=new Date(year,month,0).getDate(),me=`${year}-${String(month).padStart(2,'0')}-${last}`;
  return start<=me&&end>=ms;
}

function S15_cityCountryMap(){
  const out={};
  trips().forEach(t=>{const cs=(t.countries||inferCountries(t)).filter(Boolean);if(cs.length===1)getCities(t).forEach(c=>out[c]=cs[0])});
  Object.values(S15_FLIGHT().airports||{}).forEach(a=>{if(a.city&&a.country)out[a.city]=a.country});
  return out;
}
function S15_nearestCountry(lat,lng){
  let best=null,dist=Infinity;
  Object.values(COUNTRY_META).forEach(c=>{if(c.lat==null||c.lng==null)return;const d=S15_haversine({lat,lng},{lat:c.lat,lng:c.lng});if(d<dist){dist=d;best=c.name}});
  return best||'';
}
function S15_nearestCity(lat,lng){
  let best='',dist=Infinity;
  Object.entries(CITY_GEO).forEach(([city,g])=>{const d=S15_haversine({lat,lng},{lat:g[0],lng:g[1]});if(d<dist){dist=d;best=city}});
  return dist<=200?best:'';
}
function S15_countryCities(name,related){
  const map=S15_cityCountryMap(),out=new Set();
  related.forEach(t=>getCities(t).forEach(c=>{
    let country=map[c];
    if(!country&&CITY_GEO[c])country=S15_nearestCountry(CITY_GEO[c][0],CITY_GEO[c][1]);
    if(country===name)out.add(c);
  }));
  return [...out];
}

function S15_recordedCitySegments(){
  const seg=[];
  (typeof historicalTrips==='function'?historicalTrips():trips()).forEach(t=>{
    const cities=getCities(t).filter(c=>CITY_GEO[c]);
    for(let i=1;i<cities.length;i++){
      const a=CITY_GEO[cities[i-1]],b=CITY_GEO[cities[i]];
      seg.push({from:{lat:a[0],lng:a[1]},to:{lat:b[0],lng:b[1]},kind:'trip',label:t.title});
    }
  });
  return seg;
}

function S15_clusterPhotos(items){
  const sorted=items.filter(x=>x.date).sort((a,b)=>a.date-b.date),groups=[];
  sorted.forEach(x=>{
    const prev=groups.at(-1)?.at(-1),gap=prev?(x.date-prev.date)/86400000:Infinity;
    if(!groups.length||gap>3)groups.push([x]);else groups.at(-1).push(x);
  });
  return groups.filter(g=>g.length);
}
function S15_candidateFromGroup(group,index){
  const cityCountry=S15_cityCountryMap(),cities={},countries={};
  group.forEach(x=>{
    if(x.city)cities[x.city]=(cities[x.city]||0)+1;
    const c=x.city&&cityCountry[x.city]?cityCountry[x.city]:x.country;
    if(c)countries[c]=(countries[c]||0)+1;
  });
  const pick=o=>Object.entries(o).sort((a,b)=>b[1]-a[1])[0]?.[0]||'';
  const start=group[0].date.toISOString().slice(0,10),end=group.at(-1).date.toISOString().slice(0,10);
  return {id:`candidate-${index}`,start,end,city:pick(cities),country:pick(countries),cities:Object.keys(cities),items:group};
}
async function S15_parsePhotoFiles(files){
  const out=[];
  for(const f of files){
    let ex=null;try{ex=await exifr.parse(f,{gps:true,tiff:true,exif:true})}catch{}
    const date=ex?.DateTimeOriginal||ex?.CreateDate||new Date(f.lastModified||0);
    const validDate=date&&Number.isFinite(new Date(date).getTime());if(!validDate)continue;
    const lat=Number(ex?.latitude),lng=Number(ex?.longitude),hasGps=Number.isFinite(lat)&&Number.isFinite(lng);
    const city=hasGps?S15_nearestCity(lat,lng):'',country=city&&S15_cityCountryMap()[city]?S15_cityCountryMap()[city]:(hasGps?S15_nearestCountry(lat,lng):'');
    out.push({file:f,date:new Date(date),lat:hasGps?lat:null,lng:hasGps?lng:null,city,country});
  }
  return out;
}
