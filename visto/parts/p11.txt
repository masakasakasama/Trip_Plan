function haversineKm(a,b,c,d){const R=6371,toR=x=>x*Math.PI/180,da=toR(c-a),db=toR(d-b),q=Math.sin(da/2)**2+Math.cos(toR(a))*Math.cos(toR(c))*Math.sin(db/2)**2;return 2*R*Math.asin(Math.sqrt(q))}
