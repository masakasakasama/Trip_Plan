
/* Persona / cloud-first / theme extension */
function S17_persona(){return window.TRIP_OS_PERSONA==='rebecca'?'rebecca':'tatsu'}
function S17_profileDefaults(id){return id==='rebecca'?{name:'Rebecca',theme:'cute'}:{name:'Tatsu',theme:'midnight'}}
function S17_tripParticipants(t){
  if(Array.isArray(t?.participants)&&t.participants.length){
    const clean=[...new Set(t.participants.map(x=>String(x).toLowerCase()).filter(x=>x==='tatsu'||x==='rebecca'))];
    if(clean.length)return clean;
  }
  const names=(t?.travelers||[]).map(x=>String(x).toLowerCase());
  const hasRebecca=names.some(x=>x.includes('rebecca'));
  const hasTatsu=names.some(x=>x.includes('tatsu')||x.includes('夫')||x.includes('達也'));
  if(hasRebecca&&hasTatsu)return ['tatsu','rebecca'];
  if(hasRebecca&&!hasTatsu)return ['rebecca'];
  return ['tatsu'];
}
function S17_activeProfile(){
  const id=S17_persona();
  state.travelOS=state.travelOS||{};
  state.travelOS.profiles=state.travelOS.profiles||{};
  state.travelOS.profiles[id]=state.travelOS.profiles[id]||S17_profileDefaults(id);
  return state.travelOS.profiles[id];
}
function S17_applyTheme(){
  const p=S17_activeProfile(),theme=p.theme||S17_profileDefaults(S17_persona()).theme;
  document.body.dataset.theme=theme;
  document.documentElement.dataset.theme=theme;
}
function trips(){
  const all=Array.isArray(state.base?.trips)?state.base.trips:[];
  const id=S17_persona();
  return all.filter(t=>S17_tripParticipants(t).includes(id));
}
function normalize(){
  state.base=state.base&&Array.isArray(state.base.trips)?state.base:{schemaVersion:4,activeTripId:'',trips:[]};
  mergeHistorySeed();
  state.base.schemaVersion=Math.max(4,Number(state.base.schemaVersion||0));
  state.base.trips.forEach(t=>{if(!Array.isArray(t.participants)||!t.participants.length)t.participants=S17_tripParticipants(t)});
  state.travelOS=state.base.travelOS||state.travelOS||{};
  state.travelOS.profiles=state.travelOS.profiles||{};
  if(!state.travelOS.profiles.tatsu)state.travelOS.profiles.tatsu=Object.assign(S17_profileDefaults('tatsu'),cloneData(state.travelOS.profile||{}));
  if(!state.travelOS.profiles.rebecca)state.travelOS.profiles.rebecca=S17_profileDefaults('rebecca');
  state.travelOS.bucketsByProfile=state.travelOS.bucketsByProfile||{tatsu:Array.isArray(state.travelOS.bucket)?state.travelOS.bucket:[],rebecca:[]};
  if(!Array.isArray(state.travelOS.bucketsByProfile.tatsu))state.travelOS.bucketsByProfile.tatsu=[];
  if(!Array.isArray(state.travelOS.bucketsByProfile.rebecca))state.travelOS.bucketsByProfile.rebecca=[];
  state.travelOS.photoMetaByProfile=state.travelOS.photoMetaByProfile||{tatsu:Array.isArray(state.travelOS.photoMeta)?state.travelOS.photoMeta:[],rebecca:[]};
  if(!Array.isArray(state.travelOS.photoMetaByProfile.tatsu))state.travelOS.photoMetaByProfile.tatsu=[];
  if(!Array.isArray(state.travelOS.photoMetaByProfile.rebecca))state.travelOS.photoMetaByProfile.rebecca=[];
  const id=S17_persona();
  state.travelOS.profile=state.travelOS.profiles[id];
  state.travelOS.bucket=state.travelOS.bucketsByProfile[id];
  state.travelOS.photoMeta=state.travelOS.photoMetaByProfile[id];
  state.travelOS.settings=state.travelOS.settings||{};
  state.travelOS.settings.cloudSync=true;
  state.base.travelOS=state.travelOS;
  const visible=trips();
  if(selectedTripId&&!visible.some(t=>t.id===selectedTripId))selectedTripId='';
  if(!selectedTripId){
    const active=visible.find(t=>t.id===state.base.activeTripId);
    selectedTripId=active?.id||visible[0]?.id||'';
  }
  S17_applyTheme();
}

var S17_saveTimer=null,S17_pushBusy=false,S17_pushAgain=false;
function loadLocal(){
  try{localStorage.removeItem('trip-os-demo-v1')}catch{}
  state.base=null;
  normalize();
}
async function seedFromCloud(){
  try{
    const r=await fetch(WORKER,{cache:'no-store'});if(!r.ok)throw new Error(String(r.status));
    state.base=await r.json();
    remoteSha=(r.headers.get('X-Trip-Sha')||r.headers.get('ETag')||'').replaceAll('"','');
    normalize();
    const badge=$('#syncBadge');if(badge)badge.textContent='cloud synced';
  }catch(e){
    state.base={schemaVersion:4,activeTripId:'',trips:[],travelOS:{profiles:{tatsu:S17_profileDefaults('tatsu'),rebecca:S17_profileDefaults('rebecca')},bucketsByProfile:{tatsu:[],rebecca:[]},photoMetaByProfile:{tatsu:[],rebecca:[]},settings:{cloudSync:false}}};
    normalize();
    const badge=$('#syncBadge');if(badge)badge.textContent='cloud offline';
  }
}
function saveLocal(){
  normalize();
  const badge=$('#syncBadge');if(badge)badge.textContent='saving…';
  clearTimeout(S17_saveTimer);
  S17_saveTimer=setTimeout(()=>pushCloud(),450);
}
function S17_mergePersonaState(remote,local,persona){
  const out=cloneData(remote||{schemaVersion:4,trips:[]});
  out.trips=Array.isArray(out.trips)?out.trips:[];
  const localTrips=(local?.trips||[]).filter(t=>S17_tripParticipants(t).includes(persona));
  const localIds=new Set(localTrips.map(t=>t.id));
  out.trips=out.trips.filter(t=>!localIds.has(t.id)&&!S17_tripParticipants(t).every(p=>p===persona));
  localTrips.forEach(t=>out.trips.push(cloneData(t)));
  out.travelOS=out.travelOS||{};
  const lto=local?.travelOS||{};
  out.travelOS.profiles=out.travelOS.profiles||{};
  if(lto.profiles?.[persona])out.travelOS.profiles[persona]=cloneData(lto.profiles[persona]);
  out.travelOS.bucketsByProfile=out.travelOS.bucketsByProfile||{};
  if(lto.bucketsByProfile?.[persona])out.travelOS.bucketsByProfile[persona]=cloneData(lto.bucketsByProfile[persona]);
  out.travelOS.photoMetaByProfile=out.travelOS.photoMetaByProfile||{};
  if(lto.photoMetaByProfile?.[persona])out.travelOS.photoMetaByProfile[persona]=cloneData(lto.photoMetaByProfile[persona]);
  return out;
}
async function pullCloud(){await seedFromCloud();toast('クラウドから更新');render()}
async function pushCloud(){
  if(S17_pushBusy){S17_pushAgain=true;return}
  S17_pushBusy=true;
  try{
    normalize();
    if(!remoteSha){
      const first=await fetch(WORKER,{cache:'no-store'});if(!first.ok)throw new Error('preload '+first.status);
      remoteSha=(first.headers.get('X-Trip-Sha')||first.headers.get('ETag')||'').replaceAll('"','');
    }
    let body=cloneData(state.base),attempt=0;
    while(attempt<2){
      const r=await fetch(WORKER,{method:'PUT',headers:{'Content-Type':'application/json','If-Match':remoteSha},body:JSON.stringify(body)});
      if(r.status===409){
        const j=await r.json();remoteSha=j.sha||remoteSha;body=S17_mergePersonaState(j.state,body,S17_persona());attempt++;continue;
      }
      if(!r.ok)throw new Error(String(r.status));
      const j=await r.json();remoteSha=j.sha||remoteSha;state.base=body;normalize();
      const badge=$('#syncBadge');if(badge)badge.textContent='cloud saved';
      return;
    }
    throw new Error('conflict');
  }catch(e){
    const badge=$('#syncBadge');if(badge)badge.textContent='save failed';
    toast('クラウド保存失敗 '+e.message);
  }finally{
    S17_pushBusy=false;
    if(S17_pushAgain){S17_pushAgain=false;setTimeout(()=>pushCloud(),80)}
  }
}

/* Globe: rotation + zoom. Points on the far side are intentionally almost transparent. */
drawGlobe=async function(){
  const svg=d3.select('#globe');if(svg.empty())return;
  const node=svg.node(),w=node.clientWidth||900,h=node.clientHeight||650;
  svg.attr('viewBox',`0 0 ${w} ${h}`).selectAll('*').remove();
  const baseScale=Math.min(w,h)*.38;
  const projection=d3.geoOrthographic().translate([w/2,h/2+20]).scale(baseScale).clipAngle(90).rotate([-135,25]);
  const path=d3.geoPath(projection),data=await ensureWorldData();
  svg.append('circle').attr('class','globe-sphere').attr('cx',w/2).attr('cy',h/2+20).attr('r',baseScale);
  const visitedIds=new Set(trips().flatMap(tripCountryIds)),bucketNames=new Set(state.travelOS.bucket.filter(b=>!b.done).map(b=>String(b.country||b.name).toLowerCase()));
  const g=svg.append('g');
  const countries=g.selectAll('path.globe-country').data(data).join('path').attr('class',d=>`globe-country ${visitedIds.has(String(d.id).padStart(3,'0'))?'visited':bucketNames.has(String(d.properties.name).toLowerCase())?'bucket':''}`).attr('d',path).on('click',(e,d)=>showCountry(String(d.id).padStart(3,'0'),d.properties.name));
  const routes=[];
  trips().filter(t=>!t.archived).forEach(t=>{const pts=getCities(t).map(c=>CITY_GEO[c]).filter(Boolean).map(x=>[x[1],x[0]]);if(pts.length>1)routes.push({t,geo:{type:'LineString',coordinates:pts}})});
  const routePaths=g.selectAll('path.globe-route').data(routes).join('path').attr('class','globe-route').attr('d',d=>path(d.geo));
  const cities=[];
  trips().filter(t=>!t.archived).forEach(t=>getCities(t).forEach(c=>{const x=CITY_GEO[c];if(x)cities.push({t,city:c,coord:[x[1],x[0]]})}));
  const cityDots=g.selectAll('circle.globe-city').data(cities).join('circle').attr('class','globe-city').on('click',(e,d)=>{e.stopPropagation();selectedTripId=d.t.id;navigate('trips')});
  function isFront(coord){const r=projection.rotate(),center=[-r[0],-r[1]];return d3.geoDistance(coord,center)<=Math.PI/2}
  function redraw(){
    svg.select('.globe-sphere').attr('r',projection.scale());
    countries.attr('d',path);routePaths.attr('d',d=>path(d.geo));
    cityDots.attr('cx',d=>projection(d.coord)?.[0]??-99).attr('cy',d=>projection(d.coord)?.[1]??-99).attr('r',d=>isFront(d.coord)?4:2).attr('opacity',d=>isFront(d.coord)?0.95:0.055).classed('backside',d=>!isFront(d.coord));
  }
  redraw();
  svg.call(d3.drag().on('drag',e=>{const r=projection.rotate();projection.rotate([r[0]+e.dx*.38,Math.max(-75,Math.min(75,r[1]-e.dy*.38))]);redraw()}));
  function setZoom(k){projection.scale(Math.max(baseScale*.62,Math.min(baseScale*2.7,k)));redraw()}
  node.addEventListener('wheel',e=>{e.preventDefault();setZoom(projection.scale()*(e.deltaY>0?0.9:1.1))},{passive:false});
  let pinch=0;
  node.addEventListener('touchstart',e=>{if(e.touches.length===2)pinch=Math.hypot(e.touches[0].clientX-e.touches[1].clientX,e.touches[0].clientY-e.touches[1].clientY)},{passive:true});
  node.addEventListener('touchmove',e=>{if(e.touches.length!==2||!pinch)return;e.preventDefault();const n=Math.hypot(e.touches[0].clientX-e.touches[1].clientX,e.touches[0].clientY-e.touches[1].clientY);setZoom(projection.scale()*(n/pinch));pinch=n},{passive:false});
  window.__tripGlobeControl={zoomIn:()=>setZoom(projection.scale()*1.2),zoomOut:()=>setZoom(projection.scale()*0.82),reset:()=>{projection.rotate([-135,25]);setZoom(baseScale)}};
};
